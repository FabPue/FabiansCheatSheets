Document Structure
<!-- Basic HTML5 Boilerplate -->
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport"
    content="width=device-width,
    initial-scale=1.0">
  <title>Page Title</title>
  <link rel="stylesheet" href="style.css">
</head>
<body>
  <!-- content -->
  <script src="app.js"></script>
</body>
</html>

Images & Media
<!-- Image -->
<img src="img.jpg"
     alt="description"
     width="400" height="300"
     loading="lazy"
     decoding="async">

<!-- Responsive image -->
<img src="small.jpg"
     srcset="sm.jpg 480w,
             md.jpg 800w,
             lg.jpg 1200w"
     sizes="(max-width:600px)
            480px, 800px"
     alt="responsive">

<!-- Picture element -->
<picture>
  <source media="(min-width:800px)"
          srcset="large.webp"
          type="image/webp">
  <img src="fallback.jpg" alt="img">
</picture>
<!-- Video -->
<video controls autoplay muted
       loop width="640"
       poster="thumb.jpg">
  <source src="vid.mp4"
          type="video/mp4">
  <source src="vid.webm"
          type="video/webm">
  <track kind="subtitles"
         src="sub.vtt"
         srclang="en"
         label="English" default>
</video>

<!-- Audio -->
<audio controls preload="metadata">
  <source src="audio.mp3"
          type="audio/mpeg">
</audio>

<!-- Embed / iFrame -->
<iframe src="https://..."
        width="600" height="400"
        title="Embedded"
        loading="lazy"
        sandbox="allow-scripts">
</iframe>

Form Attributes & Validation
<!-- Common input attrs -->
name="fieldName"       form key
id="fieldId"          unique id
value="default"       default value
placeholder="hint"   ghost text
required              must be filled
disabled              not interactive
readonly              not editable
autofocus             auto-focus
autocomplete="off"
tabindex="1"
form="formId"         link to form

<!-- Validation attrs -->
minlength="3"
maxlength="50"
min="0" max="100"
pattern="[A-Za-z]{3}"
multiple              multiple values

<!-- Fieldset / Legend -->
<fieldset>
  <legend>Group</legend>
  <input type="text">
</fieldset>

<!-- Progress / Meter -->
<progress value="70" max="100"></progress>
<meter value="0.6" min="0" max="1"
       low="0.2" high="0.8"
       optimum="0.9"></meter>
<output for="range">50</output>

Interactive, Script & Miscellaneous Elements
<!-- Details / Summary -->
<details open>
  <summary>Click to expand</summary>
  <p>Hidden content here.</p>
</details>

<!-- Dialog -->
<dialog id="myDialog">
  <p>Modal content</p>
  <button
    onclick="this.closest('dialog')
      .close()">Close</button>
</dialog>
// JS:
document.getElementById('myDialog')
  .showModal();

<!-- Template -->
<template id="tmpl">
  <li class="item"></li>
</template>
// JS:
const t = document
  .getElementById('tmpl').content;
const clone = t.cloneNode(true);
<!-- Script loading -->
<script src="app.js"></script>
<!-- defer: after DOM -->
<script defer src="app.js"></script>
<!-- async: parallel load -->
<script async src="app.js"></script>
<!-- module -->
<script type="module"
        src="app.mjs"></script>
<!-- importmap -->
<script type="importmap">
{
  "imports": {
    "lodash": "/lib/lodash.js"
  }
}
</script>
